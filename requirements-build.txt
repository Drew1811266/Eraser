# Build-time tools. None of these end up in the shipped application.

#   pip install -r requirements.txt -r requirements-build.txt

# cudart land in nvidia/cu13/bin/x86_64, cuDNN in nvidia/cudnn/bin.
nvidia-cuda-runtime==13.5.37
nvidia-cublas==13.10.0.20
nvidia-cudnn-cu13==9.28.1.4
